cmake_minimum_required(VERSION 3.16)
project(BARR VERSION 0.4.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(Threads REQUIRED)
option(BARR_SANITIZE "Enable address/undefined sanitizers" OFF)
add_library(barr INTERFACE)
target_include_directories(barr INTERFACE include)
target_link_libraries(barr INTERFACE Threads::Threads)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(barr INTERFACE -Wall -Wextra -Wpedantic)
  if(BARR_SANITIZE)
    target_compile_options(barr INTERFACE -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(barr INTERFACE -fsanitize=address,undefined)
  endif()
endif()
add_executable(barr_solver src/main.cpp)
target_link_libraries(barr_solver PRIVATE barr)
add_executable(barr_tests tests/test_core.cpp)
target_link_libraries(barr_tests PRIVATE barr)
enable_testing()
add_test(NAME core_properties COMMAND barr_tests)
add_executable(barr_probe src/probe.cpp)
target_link_libraries(barr_probe PRIVATE barr)
add_executable(barr_coupling_tests tests/test_coupling.cpp)
target_link_libraries(barr_coupling_tests PRIVATE barr)
add_test(NAME coupling_properties COMMAND barr_coupling_tests)

add_executable(barr_selective_tests tests/test_selective.cpp)
target_link_libraries(barr_selective_tests PRIVATE barr)
add_test(NAME selective_properties COMMAND barr_selective_tests)

add_executable(barr_sketch_oracle src/sketch_oracle.cpp)
target_link_libraries(barr_sketch_oracle PRIVATE barr)

add_executable(barr_pair_tests tests/test_pair.cpp)
target_link_libraries(barr_pair_tests PRIVATE barr)
add_test(NAME pair_properties COMMAND barr_pair_tests)
add_executable(barr_pair_oracle src/pair_oracle.cpp)
target_link_libraries(barr_pair_oracle PRIVATE barr)

add_executable(barr_pulse_ablation_tests tests/test_pulse_ablation.cpp)
target_link_libraries(barr_pulse_ablation_tests PRIVATE barr)
# Private to this unit target. Solver, probe, oracles and other tests keep the
# original production steady_clock branch and actual wall deadlines.
target_compile_definitions(barr_pulse_ablation_tests PRIVATE BARR_TEST_FIXED_CLOCK=1)
add_test(NAME pulse_ablation_properties COMMAND barr_pulse_ablation_tests)
